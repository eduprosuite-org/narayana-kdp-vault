\chapter{Unit 5: Sampling Distributions}

\section{Essential Concepts \& Cheat Sheet}

A \textbf{sampling distribution} is the probability distribution of a statistic obtained from all possible samples of size $n$ drawn from the same population.

\begin{formulabox}[Sampling Distributions Formulas]
\textbf{1. Sampling Distribution of Sample Proportion ($\hat{p}$):}
\begin{itemize}
    \item Mean: $\mu_{\hat{p}} = p$ (Unbiased estimator)
    \item Standard Deviation: $\sigma_{\hat{p}} = \sqrt{\frac{p(1-p)}{n}}$ (Valid if $n \le 0.10N$, the 10\% condition)
    \item Shape: Approximately Normal if Large Counts condition holds:
    \[
    np \ge 10 \quad \text{and} \quad n(1-p) \ge 10
    \]
\end{itemize}

\textbf{2. Sampling Distribution of Sample Mean ($\bar{x}$):}
\begin{itemize}
    \item Mean: $\mu_{\bar{x}} = \mu$ (Unbiased estimator)
    \item Standard Deviation: $\sigma_{\bar{x}} = \frac{\sigma}{\sqrt{n}}$ (Valid if $n \le 0.10N$, the 10\% condition)
    \item Shape: Approximately Normal if:
    \begin{itemize}
        \item The population distribution is Normal, \textbf{OR}
        \item \textbf{Central Limit Theorem (CLT):} Sample size is large enough ($n \ge 30$).
    \end{itemize}
\end{itemize}
\end{formulabox}

\section{The Central Limit Theorem (CLT) --- The Score-5 Truth}

\begin{trapbox}[AP Exam Trap Alert: What the CLT Actually Says]
\textbf{The Central Limit Theorem states ONLY that the SAMPLING DISTRIBUTION of $\bar{x}$ becomes approximately Normal as $n$ increases ($n \ge 30$), REGARDLESS of the population shape.}

\textbf{What the CLT DOES NOT DO:}
\begin{enumerate}
    \item It does NOT make the population normal.
    \item It does NOT make the sample data normal.
    \item It does NOT apply to proportions ($\hat{p}$). (Proportions rely on the Large Counts condition).
\end{enumerate}
\end{trapbox}
