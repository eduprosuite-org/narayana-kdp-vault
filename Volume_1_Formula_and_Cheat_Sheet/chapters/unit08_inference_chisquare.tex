\section{Essential Conditions for All Chi-Square Tests}

\begin{enumerate}
    \item \textbf{Random:} Random sample(s) or randomized experiment.
    \item \textbf{10\% Condition:} $n \le 0.10N$ (when sampling without replacement).
    \item \textbf{Large Expected Counts:} \textbf{ALL expected counts $\ge 5$}. (Expected counts must be written explicitly in a table on FRQs).
\end{enumerate}

\section{Formulas \& Degrees of Freedom}

\begin{formulabox}[Chi-Square Test Formulas]
\textbf{Chi-Square Test Statistic ($\chi^2$):}
\[
\chi^2 = \sum \frac{(\text{Observed} - \text{Expected})^2}{\text{Expected}} = \sum \frac{(O - E)^2}{E}
\]

\textbf{Expected Counts for Two-Way Tables (Homogeneity / Independence):}
\[
\text{Expected Count} = \frac{(\text{Row Total}) \times (\text{Column Total})}{\text{Table Total}}
\]

\textbf{Degrees of Freedom ($\text{df}$):}
\begin{itemize}
    \item \textbf{GOF Test:} $\text{df} = k - 1$ (where $k$ is the number of categories).
    \item \textbf{Two-Way Table (Homogeneity / Independence):} $\text{df} = (\text{rows} - 1) \times (\text{columns} - 1)$.
\end{itemize}
\end{formulabox}

\begin{trapbox}[AP Exam Trap Alert: Observed vs Expected Counts in Conditions]
Never check whether \emph{observed} counts are $\ge 5$. The condition applies strictly to \textbf{EXPECTED counts} ($E \ge 5$). Graders will deduct points if you verify observed data rather than calculated expected values!
\end{trapbox}
